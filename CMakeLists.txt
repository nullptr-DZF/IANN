cmake_minimum_required(VERSION 3.10)
project(IANN LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(OpenMP REQUIRED)
add_compile_options(-march=native)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
include_directories(${CMAKE_CURRENT_SOURCE_DIR})
function(experiment name source)
  add_executable(${name} whynot/cpp/${source} whynot/cpp/VectorDataLoader.cpp)
  target_link_libraries(${name} OpenMP::OpenMP_CXX)
endfunction()
add_executable(experiment whynot/cpp/Experiment.cpp)
target_link_libraries(experiment OpenMP::OpenMP_CXX)
experiment(expected_generate GenerateExpectedDatasets.cpp)
experiment(task_expected_generate GenerateTaskExpected.cpp)
experiment(pure_path_search PurePathSearch.cpp)
experiment(index_reconstruction IndexReconstruction.cpp)
experiment(xr_end_to_end XREndToEnd.cpp)
